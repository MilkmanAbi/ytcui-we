cmake_minimum_required(VERSION 3.16)
project(ytcui_we VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(YTWE_WERROR "Treat compiler warnings as errors" OFF)
option(YTWE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
add_library(pugixml STATIC vendor/pugixml/pugixml.cpp)
target_include_directories(pugixml PUBLIC vendor/pugixml)
add_library(ytwe_dependencies INTERFACE)
add_library(ytwe::core ALIAS ytwe_dependencies)
target_include_directories(ytwe_dependencies INTERFACE include)
target_link_libraries(ytwe_dependencies INTERFACE OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB Threads::Threads CURL::libcurl pugixml)
target_compile_options(ytwe_dependencies INTERFACE -Wall -Wextra)
if(YTWE_WERROR)
  target_compile_options(ytwe_dependencies INTERFACE -Werror)
endif()
if(YTWE_SANITIZERS)
  target_compile_options(ytwe_dependencies INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(ytwe_dependencies INTERFACE -fsanitize=address,undefined)
endif()
if(APPLE)
  target_compile_definitions(ytwe_dependencies INTERFACE YTFAST_MACOS)
elseif(CMAKE_SYSTEM_NAME MATCHES "FreeBSD|OpenBSD|NetBSD")
  target_compile_definitions(ytwe_dependencies INTERFACE YTFAST_BSD)
endif()
add_executable(ytcui-we cli/ytcui-we.cpp)
target_link_libraries(ytcui-we PRIVATE ytwe_dependencies)
target_compile_options(ytcui-we PRIVATE -Wall -Wextra)
if(WIN32)
  target_link_libraries(ytcui-we PRIVATE shell32)
endif()
add_custom_command(TARGET ytcui-we POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/vendor/ejs $<TARGET_FILE_DIR:ytcui-we>/vendor/ejs)
include(CTest)
if(BUILD_TESTING)
  set(TEST_SUITES yj youtube core auth)
  if(NOT WIN32)
    list(APPEND TEST_SUITES client select)
  endif()
  foreach(name ${TEST_SUITES})
    add_executable(test_${name} test/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE ytwe_dependencies)
    add_test(NAME ${name} COMMAND test_${name} ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures)
    set_tests_properties(${name} PROPERTIES TIMEOUT 60)
  endforeach()
  add_executable(test_download_native test/test_download_native.cpp)
  target_link_libraries(test_download_native PRIVATE ytwe_dependencies)
  find_program(NODE_EXECUTABLE NAMES node node.exe)
  if(NODE_EXECUTABLE)
    execute_process(COMMAND ${NODE_EXECUTABLE} -p "process.versions.node" OUTPUT_VARIABLE NODE_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
  endif()
  if(NODE_EXECUTABLE AND NODE_VERSION VERSION_GREATER_EQUAL 24)
    add_executable(test_runtime test/test_runtime.cpp)
    target_link_libraries(test_runtime PRIVATE ytwe_dependencies)
    add_test(NAME runtime COMMAND test_runtime ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/vendor/ejs)
    set_tests_properties(runtime PROPERTIES TIMEOUT 90)
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME downloads COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_network.py --build-dir ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(downloads PROPERTIES TIMEOUT 180)
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/test_cli.py $<TARGET_FILE:ytcui-we>)
    set_tests_properties(cli PROPERTIES TIMEOUT 60)
  endif()
endif()
install(TARGETS ytcui-we RUNTIME DESTINATION bin)
install(DIRECTORY vendor/ejs/ DESTINATION share/ytcui-we/ejs)
install(FILES LICENSE README.md THIRD_PARTY.md DESTINATION share/doc/ytcui-we)
